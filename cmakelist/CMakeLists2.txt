cmake_minimum_required(VERSION 3.16)
project(LuaDemo VERSION 1.0 LANGUAGES C CXX)

# ==============================================
# 关键修改：通过 Conan 查找 Lua（替代系统查找）
# ==============================================
# 触发 Conan 下载 Lua 依赖（需提前配置 conan_provider.cmake）
find_package(lua REQUIRED CONFIG)

# 验证 Conan 拉取的 Lua 信息
if(TARGET lua::lua)
    message(STATUS "✅ Found Lua (Conan): ${lua_VERSION}")
endif()

# ==============================================
# 其余部分与基础版一致（仅链接方式微调）
# ==============================================
set(SOURCES src/main.cpp)
add_executable(lua_demo ${SOURCES})

# 链接 Conan 管理的 Lua 库（lua::lua 是 Conan 生成的目标）
target_link_libraries(lua_demo PRIVATE lua::lua)
target_compile_features(lua_demo PRIVATE cxx_std_17)

# 可选：运行 Lua 脚本（Conan 包包含解释器）
find_program(LUA_EXECUTABLE lua HINTS ${lua_PREFIX}/bin)
add_custom_target(run_lua_script
        COMMAND ${LUA_EXECUTABLE} ${CMAKE_SOURCE_DIR}/src/main.lua
        DEPENDS lua_demo
        COMMENT "🚀 Running Lua script via Conan Lua"
)
