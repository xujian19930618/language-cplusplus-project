# 指定使用的cmake的最低版本, 可选非必须, 如果不加可能会有警告
cmake_minimum_required(VERSION 3.30)
# 定义项目名称和使用的编程语言
project(language_cmake_project C CXX)
# 设置变量的值
set(CMAKE_CXX_STANDARD 20)

include(FetchContent)
set(LUA_SOURCE_DIR   ${CMAKE_SOURCE_DIR}/third_party/lua/lua-src)

FetchContent_Declare(
        lua
        URL https://www.lua.org/ftp/lua-5.4.8.tar.gz
        SOURCE_DIR   ${LUA_SOURCE_DIR}
)
# ⚡ 现代用法：先下载源码，但不要自动生成库
# Lua 没有 CMakeLists.txt，不会创建目标，但会下载
FetchContent_MakeAvailable(lua)   # 这里不会自动生效，因为 lua 没有 CMakeLists.txt

# 手动添加 Lua 库
file(GLOB LUA_SOURCES ${LUA_SOURCE_DIR}/src/*.c)

list(REMOVE_ITEM LUA_SOURCES
        "${LUA_SOURCE_DIR}/src/lua.c"
        "${LUA_SOURCE_DIR}/src/luac.c")

add_library(lua STATIC ${LUA_SOURCES})
target_include_directories(lua PUBLIC ${LUA_SOURCE_DIR}/src)


file(GLOB SOURCES Src/*.cpp)
add_executable(${PROJECT_NAME} ${SOURCES})
target_link_libraries(${PROJECT_NAME} PRIVATE lua)
